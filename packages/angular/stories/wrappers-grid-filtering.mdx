# Grid Filtering

This section shows filtering options in the Angular Grid wrapper.

## Basic

Declarative filtering with avatar-rendered student column and school select filter.

### Template

```html
<sl-grid [items]="students" [noSkipLinks]="true">
  <sl-grid-filter-column
    path="fullName"
    header="Student"
    [renderer]="avatarRenderer"
    [scopedElements]="scopedElements"
    value="ma"></sl-grid-filter-column>
  <sl-grid-filter-column path="group.name" header="Group"></sl-grid-filter-column>
  <sl-grid-filter-column
    path="school.id"
    header="School"
    labelPath="school.name"
    mode="select"
    value="school-3"></sl-grid-filter-column>
</sl-grid>
```

### Props

```typescript
const avatarRenderer = (student: Student) => {
  const { firstName, infix, lastName, pictureUrl } = student;

  return html`
    <sl-avatar
      .displayName=${[firstName, infix, lastName].filter(Boolean).join(' ')}
      .pictureUrl=${pictureUrl}
      size="sm"></sl-avatar>
  `;
};

const scopedElements = { 'sl-avatar': Avatar };

// students: Student[] loaded via getStudents({ count: 40 })
// declarative defaults in template: value="ma" and value="school-3"
```

## Data source

Programmatic filtering through `ArrayListDataSource` with linked filter-column IDs.

### Template

```html
<sl-grid [dataSource]="ds" [noSkipLinks]="true">
  <sl-grid-filter-column
    id="filter-student"
    path="fullName"
    header="Student"
    [renderer]="avatarRenderer"
    [scopedElements]="scopedElements"></sl-grid-filter-column>
  <sl-grid-filter-column
    id="filter-school"
    path="school.id"
    header="School"
    labelPath="school.name"
    mode="select"></sl-grid-filter-column>
</sl-grid>
```

### Props

```typescript
const ds = new ArrayListDataSource(students, {
  filters: [
    { id: 'filter-school', by: 'school.id', value: 'school-1' },
    { id: 'filter-student', by: 'fullName', value: 'ma' }
  ]
});

// plus avatarRenderer and scopedElements from Basic
```

## Grand totals row

The grand totals row only counts the rows that match the active filters. Change a filter and the Student total updates. The Group and School columns have no calculated total, so they use the `grandTotals` input with your own text, here the number of distinct groups and schools in all the data. The grid does not recalculate this text when the filters change.

### Template

```html
<sl-grid [items]="students" [noSkipLinks]="true">
  <sl-grid-filter-column
    path="fullName"
    header="Student"
    value="ma"
    grand-totals-label="students"
    grand-totals-type="text"></sl-grid-filter-column>
  <sl-grid-filter-column
    path="group.name"
    header="Group"
    [grandTotals]="groups"></sl-grid-filter-column>
  <sl-grid-filter-column
    path="school.id"
    header="School"
    labelPath="school.name"
    mode="select"
    [grandTotals]="schools"></sl-grid-filter-column>
</sl-grid>
```

### Props

```typescript
{
  groups: string; // for example "5 groups"
  schools: string; // for example "3 schools"
  students: Student[];
}
```
