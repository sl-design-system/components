# Grid Pagination

This section shows pagination patterns in the Angular Grid wrapper.

## Basic

Connect a grid and paginator components to one `ArrayListDataSource` with pagination enabled.

### Template

```html
<style>
  .pagination {
    display: flex;
    gap: 1rem;
    align-items: center;
    margin-block: 1rem;
    justify-content: space-between;
  }
  sl-paginator {
    flex: 1;
    justify-content: center;
  }
</style>
<sl-grid [dataSource]="ds" [noSkipLinks]="true">
  <sl-grid-column path="id" header="Nr."></sl-grid-column>
  <sl-grid-column path="fullName" header="Student"></sl-grid-column>
  <sl-grid-column path="school.name" header="School"></sl-grid-column>
</sl-grid>
<div class="pagination">
  <sl-paginator-status [dataSource]="ds" itemLabel="students"></sl-paginator-status>
  <sl-paginator [dataSource]="ds"></sl-paginator>
  <sl-paginator-page-size
    [dataSource]="ds"
    itemLabel="Students"
    [pageSizes]="pageSizes"></sl-paginator-page-size>
</div>
```

### Props

```typescript
const ds = new ArrayListDataSource(students, {
  pagination: true,
  page: 0,
  pageSize: 10
});
const pageSizes = [5, 10, 15, 20];

// students: Student[] loaded via getStudents({ count: 50 })
```

## Filtering

Combine filter columns with paginator against the same data source.

### Template

```html
<style>
  .pagination {
    display: flex;
    gap: 1rem;
    align-items: center;
    margin-block: 1rem;
    justify-content: space-between;
  }
  sl-paginator {
    flex: 1;
    justify-content: center;
  }
</style>
<sl-grid [dataSource]="ds" [noSkipLinks]="true">
  <sl-grid-filter-column path="fullName" header="Student"></sl-grid-filter-column>
  <sl-grid-filter-column
    path="school.name"
    header="School"
    mode="select"
    labelPath="school.name"></sl-grid-filter-column>
  <sl-grid-column path="email"></sl-grid-column>
</sl-grid>
<div class="pagination">
  <sl-paginator-status [dataSource]="ds" itemLabel="students"></sl-paginator-status>
  <sl-paginator [dataSource]="ds"></sl-paginator>
  <sl-paginator-page-size
    [dataSource]="ds"
    itemLabel="Students"
    [pageSizes]="pageSizes"></sl-paginator-page-size>
</div>
```

### Props

```typescript
const ds = new ArrayListDataSource(students, {
  filters: [
    {
      id: 'filter-school',
      by: 'school.id',
      value: 'school-1'
    }
  ],
  pagination: true,
  page: 0,
  pageSize: 10
});
const pageSizes = [5, 10, 15, 20];

// students: Student[] loaded via getStudents({ count: 50 })
```

## Grand totals row

The totals in the grand totals row are calculated for all the rows that match the active filters, across all pages. They do not change when you change the page, the page size or the sorting.

### Template

```html
<sl-grid [dataSource]="ds" [noSkipLinks]="true">
  <sl-grid-column
    header="Student"
    path="name"
    grand-totals-label="students"
    grand-totals-type="text"></sl-grid-column>
  <sl-grid-column header="Class" path="class" [grandTotals]="classes"></sl-grid-column>
  <sl-grid-column
    align="end"
    header="Grade"
    path="grade"
    grand-totals-format="average"
    grand-totals-type="number"></sl-grid-column>
</sl-grid>
<div class="pagination">
  <sl-paginator-status [dataSource]="ds" itemLabel="students"></sl-paginator-status>
  <sl-paginator [dataSource]="ds"></sl-paginator>
  <sl-paginator-page-size
    [dataSource]="ds"
    itemLabel="Students"
    [pageSizes]="pageSizes"></sl-paginator-page-size>
</div>
```

### Props

```typescript
const ds = new ArrayListDataSource(pupils, { pagination: true, pageSize: 5 });
const classes = classesGrandTotals(pupils); // for example "2 classes"
const pageSizes = [5, 10, 15];
```

## Grand totals row page total

Set `grand-totals-scope="page"` to only calculate the totals for the rows on the current page. The label then says "Page total".

### Template

```html
<sl-grid grand-totals-scope="page" [dataSource]="ds" [noSkipLinks]="true">
  <sl-grid-column
    header="Student"
    path="name"
    grand-totals-label="students"
    grand-totals-type="text"></sl-grid-column>
  <sl-grid-column header="Class" path="class" [grandTotals]="classes"></sl-grid-column>
  <sl-grid-column
    align="end"
    header="Grade"
    path="grade"
    grand-totals-format="average"
    grand-totals-type="number"></sl-grid-column>
</sl-grid>
<div class="pagination">
  <sl-paginator-status [dataSource]="ds" itemLabel="students"></sl-paginator-status>
  <sl-paginator [dataSource]="ds"></sl-paginator>
  <sl-paginator-page-size
    [dataSource]="ds"
    itemLabel="Students"
    [pageSizes]="pageSizes"></sl-paginator-page-size>
</div>
```

### Props

```typescript
const ds = new ArrayListDataSource(pupils, { pagination: true, pageSize: 5 });
const classes = classesGrandTotals(pupils); // for example "2 classes"
```
